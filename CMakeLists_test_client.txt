cmake_minimum_required(VERSION 3.10)
project(TestClient)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find required packages
find_package(Protobuf REQUIRED)

# Collect generated protobuf files
file(GLOB PROTO_SOURCES "src/message/gen/*.cc")
file(GLOB PROTO_HEADERS "src/message/gen/*.h")

# Test client executable
add_executable(test_client
    test_client.cpp
    src/net/kcp/ikcp.c
    ${PROTO_SOURCES}
)

target_include_directories(test_client PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    ${CMAKE_CURRENT_SOURCE_DIR}/src/message/gen
    ${Protobuf_INCLUDE_DIRS}
)

target_link_libraries(test_client PRIVATE
    ${Protobuf_LIBRARIES}
    pthread
)

# Set compiler flags
if(MSVC)
    target_compile_options(test_client PRIVATE /W4)
else()
    target_compile_options(test_client PRIVATE -Wall -Wextra)
endif()
